function getLastWorkoutDay() { let lastDate = ''; let lastDayId = null; const days = getDays(); if (!days || days.length === 0) return null; for (const day of days) { if (!day || !day.groups) continue; let hasData = false; let latestDate = ''; for (const g of day.groups) { for (const it of g.items) { const id = exId(day.id, it.name); const log = lastLogs[id]; if (log && log.date) { hasData = true; if (log.date > latestDate) latestDate = log.date; } } } if (hasData && latestDate > lastDate) { lastDate = latestDate; lastDayId = day.id; } } return lastDayId; }
